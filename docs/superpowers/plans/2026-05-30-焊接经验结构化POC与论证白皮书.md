```python
# weldcore/model/trajectory.py
from __future__ import annotations
from dataclasses import dataclass, field
import numpy as np

@dataclass
class TrajectorySample:
    t: float                              # s
    x: float; y: float; z: float          # mm, 焊枪末端 TCP 位置
    rx: float; ry: float; rz: float       # deg, 焊枪姿态 (euler XYZ)
    current: float | None = None          # A, 可选
    voltage: float | None = None          # V, 可选
    force: float | None = None            # N, 可选 (力控)

@dataclass
class Trajectory:
    samples: list[TrajectorySample] = field(default_factory=list)

    def __len__(self) -> int:
        return len(self.samples)

    @property
    def t(self) -> np.ndarray:
        return np.array([s.t for s in self.samples])

    @property
    def xyz(self) -> np.ndarray:
        return np.array([[s.x, s.y, s.z] for s in self.samples])

    @property
    def rpy(self) -> np.ndarray:
        return np.array([[s.rx, s.ry, s.rz] for s in self.samples])

    @classmethod
    def from_arrays(cls, t, xyz, rpy) -> "Trajectory":
        xyz = np.asarray(xyz, float); rpy = np.asarray(rpy, float)
        return cls([
            TrajectorySample(float(ti), *map(float, p), *map(float, r))
            for ti, p, r in zip(np.asarray(t, float), xyz, rpy)
        ])
```

```python
# weldcore/model/weave.py
from __future__ import annotations
from dataclasses import dataclass
from enum import Enum
import numpy as np

class WeaveType(str, Enum):
    CRESCENT = "crescent"      # 月牙形
    ZIGZAG = "zigzag"          # 锯齿形
    TRAPEZOID = "trapezoid"    # 梯形摆弧
    FIGURE8 = "figure8"        # 8字形 (Task 11 扩展)

def _crescent(phase):                    # 正弦 → 月牙
    return np.sin(phase)

def _zigzag(phase):                      # 三角波 → 锯齿
    return 2/np.pi * np.arcsin(np.sin(phase))

def _trapezoid(phase, plateau: float = 0.3):   # 削顶正弦 → 梯形(边缘停留)
    k = 1.0/(1.0-plateau)
    return np.clip(k*np.sin(phase), -1.0, 1.0)

# 可生长注册表：新增摆动模板只需在此登记
WAVEFORMS = {
    WeaveType.CRESCENT: _crescent,
    WeaveType.ZIGZAG: _zigzag,
    WeaveType.TRAPEZOID: _trapezoid,
}

@dataclass
class WeaveTemplate:
    type: WeaveType
    amplitude: float           # mm, 摆幅(半峰峰)
    frequency: float           # Hz, 摆频
    left_dwell: float = 0.0    # s, 左停留
    right_dwell: float = 0.0   # s, 右停留

    def lateral(self, phase: np.ndarray) -> np.ndarray:
        return self.amplitude * WAVEFORMS[self.type](np.asarray(phase, float))
```

```python
from __future__ import annotations
from dataclasses import dataclass
from enum import Enum
import math

class GrooveType(str, Enum):
    V = "V"; X = "X"; I = "I"; FILLET = "fillet"

@dataclass
class GrooveGeometry:
    type: GrooveType
    angle_deg: float = 0.0      # 坡口角度
    gap_mm: float = 0.0         # 根部间隙
    root_face_mm: float = 0.0   # 钝边
    depth_mm: float = 0.0       # 深度
    width_mm: float = 0.0       # 宽度

    @property
    def section_area_mm2(self) -> float:
        """V 型坡口断面填充面积近似 = 三角区 + 间隙区。"""
        tri = self.depth_mm**2 * math.tan(math.radians(self.angle_deg/2))
        return tri + self.gap_mm * self.depth_mm
```

```python
# weldcore/datagen/synth.py
from __future__ import annotations
import numpy as np
from ..model.process import WeldProcess
from ..model.trajectory import Trajectory

def synthesize(process: WeldProcess, fs: float = 100.0) -> Trajectory:
    """正向：工艺参数 → 理想轨迹。fs 采样率 Hz(100=10ms)。
    平焊直线坡口：x 沿行进方向匀速，y 为摆动侧向，z=0，姿态恒定。"""
    duration = process.length_mm / process.travel_speed
    n = int(round(duration * fs)) + 1
    t = np.arange(n) / fs
    x = process.travel_speed * t
    phase = 2*np.pi * process.weave.frequency * t
    y = process.weave.lateral(phase)
    z = np.zeros_like(t)
    xyz = np.column_stack([x, y, z])
    rpy = np.tile([process.posture.work_angle_deg,
                   process.posture.travel_angle_deg, 0.0], (n, 1))
    return Trajectory.from_arrays(t, xyz, rpy)
```

```python
# weldcore/datagen/perturb.py
from __future__ import annotations
import numpy as np
from ..model.trajectory import Trajectory

def perturb(traj: Trajectory, *, tremor_mm: float = 0.0, drift_mm: float = 0.0,
            pause_count: int = 0, pause_dur: float = 0.0, seed: int = 0) -> Trajectory:
    """注入真实人工示教扰动：
      tremor_mm  手抖(xy 高斯噪声 std)
      drift_mm   路径漂移(y 方向线性累积)
      pause_*    无效停顿(冻结位置窗口，行进暂停)——decompose 须将其与工艺停留区分。"""
    rng = np.random.default_rng(seed)
    t = traj.t.copy(); xyz = traj.xyz.copy(); rpy = traj.rpy.copy()
    n = len(t)
    dt = float(np.median(np.diff(t))) if n > 1 else 0.01
    if tremor_mm > 0:
        xyz[:, :2] += rng.normal(0.0, tremor_mm, size=(n, 2))
    if drift_mm > 0:
        xyz[:, 1] += np.linspace(0.0, drift_mm, n)
    if pause_count > 0 and pause_dur > 0:
        w = max(1, int(round(pause_dur / dt)))
        for _ in range(pause_count):
            i = int(rng.integers(0, max(1, n - w)))
            xyz[i:i+w] = xyz[i]            # 冻结位置=无效停顿
    return Trajectory.from_arrays(t, xyz, rpy)
```

```python
def extract_centerline(traj: Trajectory):
    """直线焊缝：对 xy 做 PCA 主轴拟合得焊接中心线。
    返回 (origin, direction单位向量, 行进速度mm/s, 侧向残差signed mm)。
    行进速度仅用运动样本估计 → 抗无效停顿。"""
    xy = traj.xyz[:, :2]
    origin = xy.mean(axis=0)
    _, _, vt = np.linalg.svd(xy - origin, full_matrices=False)
    direction = vt[0]
    if direction[0] < 0:
        direction = -direction
    perp = np.array([-direction[1], direction[0]])
    proj = (xy - origin) @ direction          # 沿中心线弧长
    residual = (xy - origin) @ perp           # 侧向偏移(摆动)
    dproj = np.gradient(proj, traj.t)
    active = dproj > 1e-6                      # 排除冻结停顿帧
    travel_speed = float(np.median(dproj[active])) if active.any() else 0.0
    return origin, direction, travel_speed, residual
```

```python
# weldcore/decompose/weave.py
from __future__ import annotations
import numpy as np
from ..model.weave import WAVEFORMS, WeaveType

def detect_amplitude(residual: np.ndarray) -> float:
    """鲁棒摆幅 = (P95 - P5)/2，抗离群点。"""
    return float((np.percentile(residual, 95) - np.percentile(residual, 5)) / 2)

def detect_frequency(residual: np.ndarray, t: np.ndarray) -> float:
    """FFT 主峰 = 摆频。"""
    r = residual - residual.mean()
    n = len(r); dt = float(np.median(np.diff(t)))
    spec = np.abs(np.fft.rfft(r * np.hanning(n)))
    spec[0] = 0.0
    freqs = np.fft.rfftfreq(n, dt)
    return float(freqs[np.argmax(spec)])

def _harmonic_ratio(sig: np.ndarray, t: np.ndarray, freq: float) -> float:
    """三次谐波/基波 幅值比——刻画波形形状(正弦≈0，三角≈0.11，梯形更高)。"""
    n = len(sig); dt = float(np.median(np.diff(t)))
    spec = np.abs(np.fft.rfft((sig - sig.mean()) * np.hanning(n)))
    freqs = np.fft.rfftfreq(n, dt)
    amp = lambda fk: spec[np.argmin(np.abs(freqs - fk))]
    h1 = amp(freq) or 1e-9
    return amp(3*freq) / h1

def classify_type(residual: np.ndarray, t: np.ndarray, freq: float) -> WeaveType:
    """自校准：用同频参考模板各自的谐波比作基准，取最接近者。相位无关。"""
    target = _harmonic_ratio(residual, t, freq)
    phase = 2*np.pi*freq*t
    best, best_d = None, np.inf
    for wtype, fn in WAVEFORMS.items():
        d = abs(_harmonic_ratio(fn(phase), t, freq) - target)
        if d < best_d:
            best, best_d = wtype, d
    return best
```

```python
# weldcore/decompose/engine.py
from __future__ import annotations
import numpy as np
from ..model.trajectory import Trajectory
from ..model.process import WeldProcess
from ..model.weave import WeaveTemplate
from .centerline import extract_centerline
from .weave import detect_amplitude, detect_frequency, classify_type

def decompose(traj: Trajectory) -> WeldProcess:
    """L1：拟真大师轨迹 → 结构化工艺参数(WeldProcess)。"""
    origin, direction, travel_speed, residual = extract_centerline(traj)
    t = traj.t
    freq = detect_frequency(residual, t)
    amp = detect_amplitude(residual)
    wtype = classify_type(residual, t, freq)
    weave = WeaveTemplate(type=wtype, amplitude=amp, frequency=freq)
    posture = extract_posture(traj)
    proj = (traj.xyz[:, :2] - origin) @ direction
    length = float(proj.max() - proj.min())
    return WeldProcess(travel_speed=travel_speed, weave=weave,
                       posture=posture, length_mm=length)
```

```python
from __future__ import annotations
import numpy as np
from ..model.process import WeldProcess
from ..model.trajectory import Trajectory
from ..datagen.synth import synthesize

def recompose(process: WeldProcess, fs: float = 100.0, smooth: bool = True) -> Trajectory:
    """结构化参数 → 机器人可执行轨迹。
    以工艺模型为单一真源正向再生成；smooth 时用样条插补输出连续可执行路径。
    (扩展点：重采样到控制器节拍 / 直线+圆弧+NURBS 分段拟合。)"""
    traj = synthesize(process, fs=fs)
    if not smooth:
        return traj
    from scipy.interpolate import make_interp_spline
    t, xyz = traj.t, traj.xyz
    k = min(3, len(t) - 1)
    spl = make_interp_spline(t, xyz, k=k)
    return Trajectory.from_arrays(t, spl(t), traj.rpy)
```

```python
# weldcore/metrics/roundtrip.py
from __future__ import annotations
import numpy as np
from ..model.process import WeldProcess
from ..model.trajectory import Trajectory

def trajectory_rms(a: Trajectory, b: Trajectory) -> float:
    """两轨迹重采样到公共时间网格后的位置 RMS 误差(mm)。"""
    tg = np.linspace(max(a.t[0], b.t[0]), min(a.t[-1], b.t[-1]), min(len(a), len(b)))
    ra = np.column_stack([np.interp(tg, a.t, a.xyz[:, i]) for i in range(3)])
    rb = np.column_stack([np.interp(tg, b.t, b.xyz[:, i]) for i in range(3)])
    return float(np.sqrt(np.mean(np.sum((ra - rb)**2, axis=1))))

def param_errors(true: WeldProcess, rec: WeldProcess) -> dict:
    """结构化参数恢复误差(对比 ground truth)。"""
    return {
        "travel_speed_err": abs(true.travel_speed - rec.travel_speed),
        "amplitude_err": abs(true.weave.amplitude - rec.weave.amplitude),
        "frequency_err": abs(true.weave.frequency - rec.weave.frequency),
        "type_correct": true.weave.type == rec.weave.type,
    }
```

```python
# weldcore/metrics/robustness.py
from __future__ import annotations
import numpy as np
from ..model.process import WeldProcess
from ..datagen.synth import synthesize
from ..datagen.perturb import perturb
from ..decompose.engine import decompose
from .roundtrip import param_errors

def robustness_sweep(process: WeldProcess, tremor_levels, fs: float = 100.0, seed: int = 0):
    """逐级加手抖，记录参数恢复误差。
    返回 (levels, amp_err[], freq_err[], type_ok[])。"""
    ideal = synthesize(process, fs=fs)
    amp_err, freq_err, type_ok = [], [], []
    for s in tremor_levels:
        rec = decompose(perturb(ideal, tremor_mm=s, seed=seed))
        e = param_errors(process, rec)
        amp_err.append(e["amplitude_err"]); freq_err.append(e["frequency_err"])
        type_ok.append(bool(e["type_correct"]))
    return (np.asarray(tremor_levels, float), np.asarray(amp_err),
            np.asarray(freq_err), np.asarray(type_ok, bool))

def breakdown_level(levels, errors, threshold: float) -> float:
    """误差首次超阈值的扰动级别(失效边界)；始终不超则 inf。"""
    over = np.where(np.asarray(errors) > threshold)[0]
    return float(np.asarray(levels)[over[0]]) if len(over) else float("inf")
```
